## 2. 认证机制

### 2.1 双令牌机制

- **Access Token（访问令牌）**：前端存放在 LocalStorage，用于 API 请求认证
- **Refresh Token（刷新令牌）**：存储在 http-only Cookie 中，同时服务端 Redis 存储一份

### 2.2 Token 刷新策略

1. 前端在请求拦截器中自动携带 Access Token
2. 当收到 401 响应时，自动调用 `/auth/refresh` 获取新 Access Token
3. 刷新成功后重试原请求
4. 刷新失败（Refresh Token 过期）则清除登录状态，跳转登录页

### 2.3 无感刷新实现

```typescript
// api/index.ts
request.interceptors.response.use(
  (response) => response.data,
  async (error) => {
    const originalRequest = error.config
    
    // 401 且未重试过
    if (error.response?.status === 401 && !originalRequest._retry) {
      originalRequest._retry = true
      
      try {
        // 调用刷新接口（自动携带 refresh_token cookie）
        const res = await axios.post('/auth/refresh')
        const { accessToken } = res.data.data
        
        // 更新本地存储
        localStorage.setItem('accessToken', accessToken)
        
        // 重试原请求
        originalRequest.headers.Authorization = `Bearer ${accessToken}`
        return request(originalRequest)
      } catch {
        // 刷新失败，清除登录状态
        localStorage.removeItem('accessToken')
        window.location.href = '/login'
      }
    }
    
    return Promise.reject(error)
  }
)
```

## 5. 错误处理机制

### 5.1 错误码处理

| Code | 含义 | 前端处理 |
|------|------|----------|
| 0 | 成功 | 正常处理 |
| 401 | 未授权/Token 过期 | 尝试刷新 Token，失败则跳转登录 |
| 403 | 无权限 | 提示无权限，跳转首页 |
| 404 | 资源不存在 | 显示 404 页面 |
| 500 | 服务器错误 | 显示错误提示，可重试 |

### 5.2 错误处理封装

```typescript
// utils/request.ts
export const handleError = (error: any) => {
  const status = error.response?.status
  const msg = error.response?.data?.msg || '请求失败'
  
  switch (status) {
    case 401:
      // Token 过期，尝试刷新
      break
    case 403:
      toast.error('无权访问')
      router.push('/')
      break
    case 404:
      router.push('/404')
      break
    default:
      toast.error(msg)
  }
}
```

## 6. 接口封装示例

### 6.1 文章接口

```typescript
// api/article.ts
import request from './index'
import type { Article, ArticleListParams, PageResult, ApiResponse } from '@/types'

export const getArticleList = (params?: ArticleListParams) => {
  return request.get<ApiResponse<PageResult<Article>>>('/api/article', { params })
}

export const getArticleById = (id: number) => {
  return request.get<ApiResponse<Article>>(`/api/article/${id}`)
}

export const createArticle = (data: Partial<Article>) => {
  return request.post<ApiResponse<void>>('/api/article', data)
}

export const updateArticle = (id: number, data: Partial<Article>) => {
  return request.put<ApiResponse<void>>(`/api/article/${id}`, data)
}

export const deleteArticle = (id: number) => {
  return request.delete<ApiResponse<void>>(`/api/article/${id}`)
}

export const toggleLike = (id: number, like: boolean) => {
  const url = like ? `/api/article/${id}/like` : `/api/article/${id}/unlike`
  return request.put<ApiResponse<void>>(url)
}

export const getLikeStatus = (id: number) => {
  return request.get<ApiResponse<boolean>>(`/api/article/${id}/like`)
}
```
